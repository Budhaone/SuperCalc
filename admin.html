valFinanciamento.toFixed(2) : 0; recalculateAll(); }); // Evento ao alterar o valor digitado do Financiamento document.getElementById("valFinanciamento").addEventListener("input", function() { const valorTotal = parseFloat(document.getElementById("valorTotalTabela").value) || 0; let valFinanciamento = parseFloat(this.value) || 0; // Recalcula a Entrada/Ato para fechar 100% let valAto = valorTotal - valFinanciamento; document.getElementById("valAto").value = valAto > 0 ? valAto.toFixed(2) : 0; recalculateAll(); });
